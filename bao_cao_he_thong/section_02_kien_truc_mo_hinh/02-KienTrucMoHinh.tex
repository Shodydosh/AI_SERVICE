\subsection{Kiến trúc Two-Tower Architecture}

\textbf{Theo codebase hiện tại}, hệ thống sử dụng Two-Tower Architecture với:

\subsubsection{Job Tower Encoder}

\textbf{File}: \texttt{src/embeddings/job\_tower\_encoder.py}

\textbf{Theo codebase hiện tại}, class \texttt{JobTowerEncoder}:

\begin{itemize}
    \item \textbf{Chức năng}: Encode jobs thành 3 embeddings riêng biệt:
    \begin{itemize}
        \item \texttt{title\_embedding}: Embedding cho job title
        \item \texttt{skills\_embedding}: Embedding cho job skills
        \item \texttt{requirement\_embedding}: Embedding cho job requirements
    \end{itemize}
    
    \item \textbf{Khởi tạo}:
    \begin{itemize}
        \item Parameter: \texttt{model\_name: str = None}
        \item Sử dụng model từ \texttt{settings.EMBEDDING\_MODEL} nếu không chỉ định
        \item Fallback model: \texttt{"sentence-transformers/paraphrase-multilingual-MiniLM-L12-v2"}
        \item Load model bằng \texttt{load\_embedding\_model()}
        \item Dimension: \texttt{model.get\_sentence\_embedding\_dimension()}
        \item Check Vietnamese tokenization requirement: \texttt{'SimCSE-VietNamese' in model\_name}
    \end{itemize}
    
    \item \textbf{Preprocessing functions}:
    \begin{itemize}
        \item \texttt{preprocess\_job\_title()}: 
        \begin{itemize}
            \item Translate Vietnamese → English bằng \texttt{VietnameseTranslator}
            \item Strip và lowercase
            \item Remove extra spaces
            \item Truncate 200 chars
        \end{itemize}
        \item \texttt{preprocess\_job\_skills()}: 
        \begin{itemize}
            \item Translate, normalize, truncate 1000 chars
        \end{itemize}
        \item \texttt{preprocess\_job\_requirements()}: 
        \begin{itemize}
            \item Translate, normalize
            \item Nếu có > 3 sentences và > 2000 chars: chỉ lấy 3 sentences đầu
            \item Truncate 2000 chars
        \end{itemize}
    \end{itemize}
    
    \item \textbf{Tokenization}:
    \begin{itemize}
        \item Method \texttt{\_tokenize\_vietnamese()}: Sử dụng pyvi nếu model yêu cầu
        \item Check \texttt{PYVI\_AVAILABLE} flag
        \item Fallback về original text nếu error
    \end{itemize}
    
    \item \textbf{Encoding}:
    \begin{itemize}
        \item Method \texttt{\_generate\_single\_embedding()}: 
        \begin{itemize}
            \item Nếu text empty: return zero embedding
            \item Tokenize Vietnamese nếu cần
            \item Encode bằng \texttt{model.encode()} với \texttt{normalize\_embeddings=True}
            \item Ensure normalization: \texttt{embedding / norm} nếu norm > 0
            \item Return list of floats
        \end{itemize}
        \item Method \texttt{encode\_job()}: 
        \begin{itemize}
            \item Preprocess 3 fields
            \item Generate 3 embeddings
            \item Return dict với keys: \texttt{'title\_embedding'}, \texttt{'skills\_embedding'}, \texttt{'requirement\_embedding'}
        \end{itemize}
    \end{itemize}
\end{itemize}

\subsubsection{Candidate Tower Encoder}

\textbf{File}: \texttt{src/embeddings/candidate\_tower\_encoder.py}

\textbf{Theo codebase hiện tại}, class \texttt{CandidateTowerEncoder} tương tự \texttt{JobTowerEncoder}:

\begin{itemize}
    \item \textbf{3 embeddings riêng biệt}:
    \begin{itemize}
        \item \texttt{title\_embedding}: Candidate title (desired job)
        \item \texttt{skills\_embedding}: Candidate skills
        \item \texttt{experience\_embedding}: Candidate experience
    \end{itemize}
    
    \item \textbf{Preprocessing}:
    \begin{itemize}
        \item \texttt{preprocess\_candidate\_experience()}: Translate, normalize, truncate 2000 chars
        \item Reuse \texttt{preprocess\_job\_title()} và \texttt{preprocess\_job\_skills()} từ \texttt{job\_tower\_encoder}
    \end{itemize}
    
    \item \textbf{Method \texttt{encode\_candidate()}}: Tương tự \texttt{encode\_job()} nhưng cho candidates
\end{itemize}

\textbf{Theo codebase hiện tại}, mỗi record (job hoặc candidate) có 3 embeddings riêng biệt thay vì 1 embedding duy nhất. Điều này cho phép:

\subsection{Two-Tower Model Architecture}

\textbf{Theo codebase hiện tại}, Two-Tower Model được định nghĩa trong \texttt{src/models/two\_tower\_model.py}:

\subsubsection{CandidateTower Neural Network}

\textbf{Theo codebase hiện tại}:

\begin{itemize}
    \item \textbf{Input}: 3 embeddings (title, skills, experience) - mỗi embedding 768 dimensions
    \item \textbf{Architecture}:
    \begin{itemize}
        \item Concatenate 3 embeddings → \texttt{input\_dim = 768 * 3 = 2304}
        \item Hidden layers: \texttt{[512, 256]} (configurable)
        \item Output layer: \texttt{256} dimensions (configurable)
        \item Batch normalization: \texttt{use\_batch\_norm = True} (default)
        \item Dropout: \texttt{0.1} (default)
        \item L2 normalization ở output
    \end{itemize}
    \item \textbf{Activation}: ReLU
    \item \textbf{Weight initialization}: Xavier uniform cho Linear layers, constant cho BatchNorm
    \item \textbf{Forward pass}:
    \begin{itemize}
        \item Input: \texttt{title\_emb}, \texttt{skills\_emb}, \texttt{experience\_emb} (mỗi cái \texttt{[batch\_size, 768]})
        \item Concatenate: \texttt{torch.cat([title\_emb, skills\_emb, experience\_emb], dim=1)} → \texttt{[batch\_size, 2304]}
        \item Pass qua network: \texttt{self.network(combined)}
        \item L2 normalize output: \texttt{F.normalize(output, p=2, dim=1)}
        \item Return: \texttt{[batch\_size, 256]}
    \end{itemize}
\end{itemize}

\subsubsection{JobTower Neural Network}

\textbf{Theo codebase hiện tại}, tương tự CandidateTower:

\begin{itemize}
    \item \textbf{Input}: 3 embeddings (title, skills, requirement) - mỗi embedding 768 dimensions
    \item \textbf{Architecture}: Giống CandidateTower
    \begin{itemize}
        \item Concatenate → \texttt{input\_dim = 2304}
        \item Hidden layers: \texttt{[512, 256]}
        \item Output layer: \texttt{256} dimensions
        \item Batch normalization và dropout
        \item L2 normalization ở output
    \end{itemize}
    \item \textbf{Forward pass}: Tương tự CandidateTower nhưng với \texttt{requirement\_emb} thay vì \texttt{experience\_emb}
\end{itemize}

\subsubsection{TwoTowerModel Class}

\textbf{Theo codebase hiện tại}:

\begin{itemize}
    \item \textbf{Initialization}:
    \begin{itemize}
        \item Parameters: \texttt{embedding\_dim=768}, \texttt{candidate\_hidden\_dims=[512, 256]}, \texttt{job\_hidden\_dims=[512, 256]}, \texttt{output\_dim=256}, \texttt{dropout=0.1}, \texttt{use\_batch\_norm=True}
        \item Tạo \texttt{CandidateTower} và \texttt{JobTower} instances
        \item Store \texttt{output\_dim}
    \end{itemize}
    
    \item \textbf{Forward pass}:
    \begin{itemize}
        \item Input: 6 embeddings (3 cho candidate, 3 cho job)
        \item Encode candidate qua \texttt{CandidateTower} → \texttt{candidate\_repr [batch\_size, 256]}
        \item Encode job qua \texttt{JobTower} → \texttt{job\_repr [batch\_size, 256]}
        \item Return: \texttt{(candidate\_repr, job\_repr)}
    \end{itemize}
    
    \item \textbf{Similarity computation}:
    \begin{itemize}
        \item Method \texttt{compute\_similarity()}: 
        \begin{itemize}
            \item Dot product: \texttt{torch.sum(candidate\_repr * job\_repr, dim=1)}
            \item Vì cả 2 đều L2 normalized → đây là cosine similarity
            \item Return: \texttt{[batch\_size]} similarity scores
        \end{itemize}
    \end{itemize}
    
    \item \textbf{Inference methods}:
    \begin{itemize}
        \item \texttt{encode\_candidate()}: Encode candidate vào representation
        \item \texttt{encode\_job()}: Encode job vào representation
    \end{itemize}
\end{itemize}

\textbf{Theo codebase hiện tại} (\texttt{src/services/two\_tower\_matching\_service.py}):

\subsection{Embedding Service và Caching}

\subsubsection{OptimizedEmbeddingService}

\textbf{File}: \texttt{src/services/embedding\_service.py}

\textbf{Theo codebase hiện tại}:

\begin{itemize}
    \item \textbf{Initialization}:
    \begin{itemize}
        \item Parameters: \texttt{db: Session}, \texttt{cache\_ttl\_hours: float = 12.0}, \texttt{batch\_size: int = 100}
        \item Initialize \texttt{MultiFieldEmbeddingRepository}
        \item Initialize \texttt{EmbeddingCacheManager} với TTL
        \item Lazy load encoders (chỉ load khi cần)
    \end{itemize}
    
    \item \textbf{Features}:
    \begin{itemize}
        \item Smart caching với 12-hour TTL
        \item Batch processing
        \item Content hash tracking để detect changes
        \item Database integration
    \end{itemize}
    
    \item \textbf{Method \texttt{get\_candidate\_embedding()}}:
    \begin{enumerate}
        \item Compute content hash: \texttt{MD5(title|skills|experience)}
        \item Check cache first (nếu không force refresh)
        \item Check database:
        \begin{itemize}
            \item Query \texttt{CandidateMultiEmbedding} theo \texttt{candidate\_id}
            \item Check embedding timestamp: nếu age < TTL và content hash match → use database embedding
        \end{itemize}
        \item Nếu cần compute mới:
        \begin{itemize}
            \item Encode bằng \texttt{CandidateTowerEncoder}
            \item Save to database
            \item Cache result
        \end{itemize}
        \item Return dict với 3 embeddings
    \end{enumerate}
    
    \item \textbf{Method \texttt{get\_job\_embedding()}}: Tương tự \texttt{get\_candidate\_embedding()}
    
    \item \textbf{Method \texttt{\_save\_candidate\_embedding()}}:
    \begin{itemize}
        \item Query existing candidate
        \item Nếu tồn tại: update fields và embeddings, update \texttt{embedding\_timestamp}, \texttt{content\_hash}
        \item Nếu không: create new \texttt{CandidateMultiEmbedding}
        \item Commit và log
    \end{itemize}
    
    \item \textbf{Method \texttt{\_save\_job\_embedding()}}: Tương tự cho jobs
    
    \item \textbf{Batch methods}:
    \begin{itemize}
        \item \texttt{batch\_get\_candidates\_needing\_refresh()}: Query candidates có \texttt{embedding\_timestamp < cutoff\_time}
        \item \texttt{batch\_get\_jobs\_needing\_refresh()}: Tương tự cho jobs
        \item \texttt{batch\_process\_candidates()}: Process candidates in batches
        \item \texttt{batch\_process\_jobs()}: Process jobs in batches
    \end{itemize}
\end{itemize}

\subsubsection{EmbeddingCacheManager}

\textbf{File}: \texttt{src/services/embedding\_cache\_manager.py}

\textbf{Theo codebase hiện tại}:

\begin{itemize}
    \item \textbf{Initialization}:
    \begin{itemize}
        \item Parameter: \texttt{cache\_ttl\_hours: float = 12.0}
        \item In-memory cache: \texttt{Dict[str, Dict[str, Any]]}
        \item Thread-safe: \texttt{Lock} cho operations
    \end{itemize}
    
    \item \textbf{Method \texttt{get()}}:
    \begin{enumerate}
        \item Cache key: \texttt{f"{entity\_type}:{entity\_id}"}
        \item Check cache exists
        \item Check expiration: \texttt{\_is\_expired()}
        \item Check content hash nếu provided
        \item Update \texttt{last\_accessed} timestamp
        \item Return cached embedding hoặc None
    \end{enumerate}
    
    \item \textbf{Method \texttt{set()}}:
    \begin{itemize}
        \item Store embedding với timestamp, content\_hash, metadata
        \item Thread-safe với Lock
    \end{itemize}
    
    \item \textbf{Method \texttt{invalidate()}}: Delete cache entry
    
    \item \textbf{Method \texttt{clear\_expired()}}: Clear all expired entries
    
    \item \textbf{Method \texttt{get\_stats()}}: Return cache statistics (total, valid, expired entries)
    
    \item \textbf{Method \texttt{needs\_refresh()}}:
    \begin{itemize}
        \item Check cache first
        \item Check database timestamp nếu provided
        \item Return True nếu cần refresh
    \end{itemize}
    
    \item \textbf{Global instance}: Function \texttt{get\_cache\_manager()} return singleton instance
\end{itemize}
