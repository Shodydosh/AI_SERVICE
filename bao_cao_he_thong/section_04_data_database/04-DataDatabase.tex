\section{Lớp Xử Lý Dữ Liệu}

\subsection{JDProcessor Class}

\textbf{Theo codebase hiện tại}, class \texttt{JDProcessor} được định nghĩa trong file \texttt{src/data\_processing/jd\_processor.py}.

\subsubsection{Khởi tạo}

Class được khởi tạo với tham số:
\begin{itemize}
    \item \texttt{auto\_map\_columns: bool = True}: Tự động map tên cột về định dạng chuẩn
\end{itemize}

Class có các thuộc tính:
\begin{itemize}
    \item \texttt{data: Optional[pd.DataFrame]}: DataFrame chứa dữ liệu job descriptions
    \item \texttt{auto\_map\_columns: bool}: Flag cho phép auto-map columns
\end{itemize}

\subsubsection{Phương thức load\_from\_csv()}

\textbf{Theo codebase hiện tại}, phương thức này:
\begin{enumerate}
    \item Load dữ liệu từ CSV file bằng \texttt{pd.read\_csv(file\_path)}
    \item Log số lượng job descriptions đã load
    \item Nếu \texttt{auto\_map\_columns = True}, tự động map tên cột bằng \texttt{ColumnMapper.map\_jd\_columns()}
    \item Return DataFrame
    \item Xử lý exception và log error nếu có lỗi
\end{enumerate}

\subsubsection{Phương thức load\_from\_json()}

\textbf{Theo codebase hiện tại}, phương thức này tương tự \texttt{load\_from\_csv()} nhưng:
\begin{itemize}
    \item Sử dụng \texttt{pd.read\_json(file\_path)} để load từ JSON
    \item Các bước còn lại giống như load từ CSV
\end{itemize}

\subsubsection{Phương thức validate\_data()}

\textbf{Theo codebase hiện tại}, phương thức này:
\begin{enumerate}
    \item Kiểm tra \texttt{self.data} có tồn tại không
    \item Kiểm tra các cột bắt buộc: \texttt{['job\_id', 'title', 'description']}
    \item Nếu thiếu cột bắt buộc, log error và return \texttt{False}
    \item Return \texttt{True} nếu validation thành công
\end{enumerate}

\subsubsection{Phương thức get\_field\_texts()}

\textbf{Theo codebase hiện tại}, phương thức này extract các field texts từ một row:
\begin{itemize}
    \item Helper function \texttt{get\_field(field\_name)}:
    \begin{itemize}
        \item Kiểm tra field có trong row.index không
        \item Xử lý pandas Series nếu cần
        \item Kiểm tra NaN/None
        \item Return text đã strip hoặc None
    \end{itemize}
    \item Extract các fields:
    \begin{itemize}
        \item \texttt{title}: Job title
        \item \texttt{skills}: Job skills
        \item \texttt{requirements}: Job requirements
        \item \texttt{description}: Job description (giới hạn 500 chars)
        \item \texttt{company}: Company name
        \item \texttt{location}: Job location
    \end{itemize}
    \item Return dictionary với key là field name, value là text content
\end{itemize}

\subsubsection{Phương thức get\_combined\_text()}

\textbf{Theo codebase hiện tại}, phương thức này combine các JD fields thành một text duy nhất cho embedding:
\begin{itemize}
    \item Priority order (theo codebase):
    \begin{enumerate}
        \item \textbf{Title}: Repeat 6 lần với format "Job Title: \{title\}", 4 lần với format "Position: \{title\}"
        \item \textbf{Skills}: Repeat 6 lần với format "Required Skills and Technologies: \{skills\}", 4 lần với format "Skills Required: \{skills\}"
        \item \textbf{Requirements}: Repeat 6 lần với format "Job Requirements: \{requirements\}", 4 lần với format "Required: \{requirements\}"
        \item \textbf{Description}: 1 lần với format "Job Description: \{description\}" (giới hạn 500 chars)
        \item \textbf{Company}: 1 lần với format "Company: \{company\}"
        \item \textbf{Location}: 1 lần với format "Location: \{location\}"
    \end{enumerate}
    \item Text được normalize bằng \texttt{TextEnhancer.normalize\_text()}
    \item Join các parts bằng space
    \item Fallback: nếu không có valuable fields, sử dụng bất kỳ field nào có sẵn
\end{itemize}

\subsubsection{Phương thức get\_records() và get\_record\_by\_id()}

\textbf{Theo codebase hiện tại}:
\begin{itemize}
    \item \texttt{get\_records()}: Convert DataFrame thành list of dictionaries bằng \texttt{data.to\_dict('records')}
    \item \texttt{get\_record\_by\_id(job\_id)}: Filter theo \texttt{job\_id} và return dictionary của record đầu tiên
\end{itemize}

\subsection{CandidateProcessor Class}

\textbf{Theo codebase hiện tại}, class \texttt{CandidateProcessor} được định nghĩa trong file \texttt{src/data\_processing/candidate\_processor.py}.

\subsubsection{Cấu trúc tương tự JDProcessor}

Class này có cấu trúc tương tự \texttt{JDProcessor}:
\begin{itemize}
    \item \texttt{load\_from\_csv()}: Load từ CSV với auto-map columns
    \item \texttt{load\_from\_json()}: Load từ JSON với auto-map columns
    \item \texttt{validate\_data()}: Validate với required columns là \texttt{['candidate\_id']}
\end{itemize}

\subsubsection{Phương thức get\_field\_texts()}

\textbf{Theo codebase hiện tại}, extract các fields từ candidate row:
\begin{itemize}
    \item \texttt{skills}: Candidate skills
    \item \texttt{experience} hoặc \texttt{work\_experience}: Work experience
    \item \texttt{desired\_job\_translated}: Desired job title
    \item \texttt{summary}: Professional summary
    \item \texttt{education}: Education background
    \item \texttt{industry}: Industry
    \item \texttt{workplace\_desired}: Location preference
    \item \texttt{Companies}: Previous companies
    \item \texttt{candidate\_id} hoặc \texttt{cv\_id}: Unique identifier
    \item \texttt{age}, \texttt{gender}, \texttt{desired\_salary}: Additional context
    \item \texttt{resume\_text}: Full resume (chỉ include nếu substantial)
\end{itemize}

\subsubsection{Phương thức get\_combined\_text()}

\textbf{Theo codebase hiện tại}, combine candidate fields với priority order:
\begin{enumerate}
    \item \textbf{Skills}: Repeat 6 lần "Required Skills and Technologies: \{skills\}", 4 lần "Skills: \{skills\}"
    \item \textbf{Experience}: Repeat 6 lần "Work Experience: \{experience\}", 6 lần "Job Requirements: \{experience\}"
    \item \textbf{Desired Job}: 3 lần "Job Title: \{desired\_job\}", 2 lần với format variations
    \item \textbf{Summary}: 1 lần "Professional Summary: \{summary\}"
    \item \textbf{Education}: 1 lần "Education: \{education\}"
    \item \textbf{Industry}: 1 lần "Industry: \{industry\}"
    \item \textbf{Workplace}: 1 lần "Preferred Location: \{workplace\_desired\}"
    \item \textbf{Companies}: 1 lần "Previous Companies: \{Companies\}"
    \item \textbf{Resume}: Chỉ include nếu resume text dài hơn 1.5 lần existing text
\end{enumerate}

\subsection{DataPreprocessor Class}

\textbf{Theo codebase hiện tại}, class \texttt{DataPreprocessor} được định nghĩa trong file \texttt{src/utils/data\_preprocessor.py}.

\subsubsection{Phương thức clean\_text()}

\textbf{Theo codebase hiện tại}:
\begin{enumerate}
    \item Kiểm tra NaN/None
    \item Convert sang string
    \item Remove extra whitespace: \texttt{re.sub(r'\textbackslash s+', ' ', text)}
    \item Remove special characters nhưng giữ basic punctuation: \texttt{re.sub(r'[\textasciicircum \textbackslash w\textbackslash s.,;:!?()-]', '', text)}
    \item Strip leading/trailing whitespace
    \item Return cleaned text
\end{enumerate}

\subsubsection{Phương thức normalize\_whitespace()}

\textbf{Theo codebase hiện tại}:
\begin{itemize}
    \item Replace multiple spaces, tabs, newlines với single space
    \item Pattern: \texttt{re.sub(r'\textbackslash s+', ' ', text)}
    \item Strip và return
\end{itemize}

\subsubsection{Phương thức remove\_html\_tags()}

\textbf{Theo codebase hiện tại}:
\begin{itemize}
    \item Remove HTML tags bằng regex: \texttt{re.sub(r'<[\textasciicircum >]+>', '', text)}
    \item Return text không có HTML tags
\end{itemize}

\subsubsection{Phương thức normalize\_email()}

\textbf{Theo codebase hiện tại}:
\begin{enumerate}
    \item Strip và lowercase email
    \item Validate email format bằng regex: \texttt{r'\textasciicircum [a-zA-Z0-9.\_\%+-]+@[a-zA-Z0-9.-]+\textbackslash .[a-zA-Z]\{2,\}\$'}
    \item Return normalized email hoặc None nếu invalid
    \item Log warning nếu email format không hợp lệ
\end{enumerate}

\subsubsection{Phương thức normalize\_id()}

\textbf{Theo codebase hiện tại}:
\begin{enumerate}
    \item Convert sang string và strip
    \item Remove special characters, giữ alphanumeric và underscores: \texttt{re.sub(r'[\textasciicircum \textbackslash w]', '\_', id\_str)}
    \item Return normalized ID
\end{enumerate}

\subsubsection{Phương thức extract\_and\_clean\_skills()}

\textbf{Theo codebase hiện tại}:
\begin{enumerate}
    \item Split skills text bằng delimiters: \texttt{re.split(r'[,;|]', skills)}
    \item Clean mỗi skill: strip, remove special chars
    \item Join lại bằng comma
    \item Return cleaned skills string
\end{enumerate}

    \begin{itemize}
        \item Parameters: \texttt{job\_id}, \texttt{title}, \texttt{title\_embedding}, \texttt{skills\_embedding}, \texttt{requirement\_embedding}, \texttt{skills}, \texttt{requirement}, \texttt{company}, \texttt{location}, \texttt{replace\_existing=True}
        \item Nếu \texttt{replace\_existing=True}, xóa existing job trước khi tạo mới
        \item Tạo \texttt{JobDescriptionTwoTower} object
        \item Add vào session, commit, refresh
        \item Return job object
    \end{itemize}
    \item \texttt{get\_job(job\_id)}: Query job theo \texttt{job\_id}, return \texttt{Optional[JobDescriptionTwoTower]}
    \item \texttt{get\_all\_jobs()}: Query tất cả jobs, return \texttt{List[JobDescriptionTwoTower]}
    \item \texttt{get\_jobs\_updated\_since(since\_time)}: Query jobs có \texttt{updated\_at >= since\_time}
\end{itemize}
