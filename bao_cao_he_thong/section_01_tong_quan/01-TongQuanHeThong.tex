\subsection{Giới thiệu hệ thống}

\textbf{Theo codebase hiện tại}, hệ thống AI Service là một dịch vụ khuyến nghị việc làm thông minh sử dụng kiến trúc Two-Tower Architecture để match Job Descriptions (JD) với Candidates. Hệ thống được xây dựng bằng Python với FastAPI framework, sử dụng PostgreSQL để lưu trữ dữ liệu và embeddings, FAISS cho vector search, và PyTorch cho neural network models.

\subsubsection{Entry Point}

\textbf{File}: \texttt{main.py}

\textbf{Theo codebase hiện tại}, file này khởi động FastAPI server với cấu hình từ \texttt{config/settings.py}:

\begin{verbatim}
uvicorn.run(
    "src.api.main:app",
    host=settings.API_HOST,
    port=settings.API_PORT,
    reload=True
)
\end{verbatim}

\subsubsection{Cấu hình hệ thống}

\textbf{Theo codebase hiện tại} (\texttt{config/settings.py}):

\begin{itemize}
    \item \textbf{API Host}: \texttt{0.0.0.0}
    \item \textbf{API Port}: \texttt{8000}
    \item \textbf{Embedding Model}: \texttt{VoVanPhuc/sup-SimCSE-VietNamese-phobert-base}
    \item \textbf{Embedding Dimension}: \texttt{768} (PhoBERT base)
    \item \textbf{Database}:
    \begin{itemize}
        \item \texttt{DB\_USER}: \texttt{"postgres"} (default)
        \item \texttt{DB\_PASSWORD}: \texttt{"123"} (default)
        \item \texttt{DB\_HOST}: \texttt{"localhost"}
        \item \texttt{DB\_PORT}: \texttt{5432}
        \item \texttt{DB\_NAME}: \texttt{"job\_recommendation\_db"}
    \end{itemize}
    \item \textbf{Logging}: \texttt{LOG\_LEVEL = "INFO"}
\end{itemize}

\subsubsection{Cấu hình từ Environment Variables}

\textbf{Theo codebase hiện tại}, class \texttt{Settings} sử dụng \texttt{PydanticSettings}:
\begin{itemize}
    \item Load từ file \texttt{.env}
    \item Encoding: \texttt{utf-8}
    \item Case sensitive: \texttt{True}
    \item Method \texttt{get\_database\_url()}: Construct DATABASE\_URL từ các components, URL encode password để handle special characters
\end{itemize}

\subsection{Mục tiêu và phạm vi}

\subsubsection{Chức năng chính}

\textbf{Theo codebase hiện tại}, hệ thống cung cấp các chức năng:

\begin{enumerate}
    \item \textbf{Indexing}: Index jobs và candidates với 3 embeddings riêng biệt:
    \begin{itemize}
        \item Title embedding
        \item Skills embedding
        \item Requirement/Experience embedding
    \end{itemize}
    
    \item \textbf{Search}:
    \begin{itemize}
        \item Tìm jobs cho candidates: \texttt{POST /api/v2/search/jobs}
        \item Tìm candidates cho jobs: \texttt{POST /api/v2/search/candidates}
    \end{itemize}
    
    \item \textbf{Matching}: Sử dụng Two-Tower model để tính similarity giữa candidates và jobs
    
    \item \textbf{Explainability}: Cung cấp giải thích cho kết quả matching với 5 levels (theo \texttt{src/utils/explanation\_generator.py})
\end{enumerate}

\subsection{Công nghệ sử dụng}

\textbf{Theo codebase hiện tại}, các công nghệ chính:

\subsubsection{Web Framework}

\begin{itemize}
    \item \textbf{FastAPI}: Modern web framework cho API
    \begin{itemize}
        \item Version: 2.0.0 (theo \texttt{src/api/main.py})
        \item Async support
        \item Automatic API documentation (Swagger/OpenAPI)
        \item Type validation với Pydantic
    \end{itemize}
\end{itemize}

\subsubsection{Database}

\begin{itemize}
    \item \textbf{PostgreSQL}: Relational database
    \begin{itemize}
        \item ARRAY support cho embeddings
        \item GIN indexes cho vector search
        \item Connection pooling (pool\_size=10, max\_overflow=20)
    \end{itemize}
    
    \item \textbf{SQLAlchemy}: ORM cho database operations
    \begin{itemize}
        \item Declarative models
        \item Session management
        \item Query building
    \end{itemize}
\end{itemize}

\subsubsection{Machine Learning}

\begin{itemize}
    \item \textbf{PyTorch}: Deep learning framework
    \begin{itemize}
        \item Neural network models (\texttt{src/models/two\_tower\_model.py})
        \item Training và inference
        \item Model checkpointing
    \end{itemize}
    
    \item \textbf{Sentence Transformers}: Embedding generation
    \begin{itemize}
        \item Pre-trained models
        \item Vietnamese language support
        \item Model: \texttt{VoVanPhuc/sup-SimCSE-VietNamese-phobert-base}
    \end{itemize}
\end{itemize}

\subsubsection{Vector Search}

\begin{itemize}
    \item \textbf{FAISS}: Facebook AI Similarity Search
    \begin{itemize}
        \item HNSW index cho approximate nearest neighbor search
        \item 6 indices riêng biệt (3 cho jobs, 3 cho candidates)
        \item Index persistence
    \end{itemize}
\end{itemize}

\subsubsection{Vietnamese NLP}

\begin{itemize}
    \item \textbf{pyvi}: Vietnamese tokenization
    \begin{itemize}
        \item Text preprocessing
        \item Tokenization cho Vietnamese models
        \item Optional dependency (có fallback nếu không có)
    \end{itemize}
\end{itemize}

\subsubsection{Configuration Management}

\begin{itemize}
    \item \textbf{Pydantic Settings}: Configuration management
    \begin{itemize}
        \item Environment variables
        \item Type validation
        \item Default values
    \end{itemize}
\end{itemize}
