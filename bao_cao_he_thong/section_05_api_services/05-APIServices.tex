\subsubsection{POST /api/v2/search/jobs}

\textbf{Theo codebase hiện tại}:

\begin{itemize}
    \item \textbf{Request}: \texttt{JobSearchRequest}
    \begin{itemize}
        \item \texttt{candidate\_id: str} (required)
        \item \texttt{top\_k: int = 10} (default, ge=1, le=100)
    \end{itemize}
    
    \item \textbf{Response}: \texttt{JobSearchResponse}
    \begin{itemize}
        \item \texttt{total\_matches: int}
        \item \texttt{matches: List[JobMatch]}
    \end{itemize}
    
    \item \textbf{Logic}:
    \begin{enumerate}
        \item Create \texttt{TwoTowerMatchingService(db)}
        \item Call \texttt{find\_jobs\_for\_candidate(candidate\_id, top\_k)}
        \item Convert results to \texttt{JobMatch} objects
        \item Return \texttt{JobSearchResponse}
    \end{enumerate}
    
    \item \textbf{Error handling}:
    \begin{itemize}
        \item Try-except block
        \item Log error với \texttt{exc\_info=True}
        \item Raise \texttt{HTTPException} với status 500
    \end{itemize}
\end{itemize}

\subsubsection{POST /api/v2/index/job}

\textbf{Theo codebase hiện tại}:

\begin{itemize}
    \item \textbf{Request}: \texttt{IndexJobRequest}
    \begin{itemize}
        \item \texttt{job\_id: str} (required)
        \item \texttt{title: str} (required)
        \item \texttt{skills: Optional[str] = None}
        \item \texttt{requirement: Optional[str] = None}
        \item \texttt{company: Optional[str] = None}
        \item \texttt{location: Optional[str] = None}
    \end{itemize}
    
    \item \textbf{Response}: \texttt{IndexResponse}
    \begin{itemize}
        \item \texttt{status: str} ("success")
        \item \texttt{job\_id: str}
        \item \texttt{message: str} ("Job indexed successfully")
    \end{itemize}
    
    \item \textbf{Logic}:
    \begin{enumerate}
        \item Create \texttt{JobTowerEncoder()}

        \item Create \texttt{TwoTowerRepository(db)}
        \item Save to database: \texttt{repository.create\_job()} với 3 embeddings
        \item Return success response
    \end{enumerate}
\end{itemize}

\subsubsection{GET /api/v2/health}

\textbf{Theo codebase hiện tại}:

\begin{itemize}
    \item \textbf{Response}: \texttt{HealthResponse}
    \begin{itemize}
        \item \texttt{status: str} ("healthy")
        \item \texttt{version: str} ("2.0.0")
        \item \texttt{indices: Dict[str, str]}: Status của 6 FAISS indices
        \item \texttt{database: str} ("connected")
        \item \texttt{total\_jobs: int}
        \item \texttt{total\_candidates: int}
    \end{itemize}
    
    \item \textbf{Logic}:
    \begin{enumerate}
        \item Create \texttt{TwoTowerRepository(db)}
        \item Count jobs: \texttt{len(repository.get\_all\_jobs())}
        \item Count candidates: \texttt{len(repository.get\_all\_candidates())}
        \item Check FAISS indices:
        \begin{itemize}
            \item Path: \texttt{"indices/two\_tower"}
            \item Check 6 index files:
            \begin{itemize}
                \item \texttt{job\_title\_index.faiss}
                \item \texttt{job\_skills\_index.faiss}
                \item \texttt{job\_requirement\_index.faiss}
                \item \texttt{candidate\_title\_index.faiss}
                \item \texttt{candidate\_skills\_index.faiss}
                \item \texttt{candidate\_experience\_index.faiss}
            \end{itemize}
            \item Status: "loaded" nếu file exists, "not\_found" nếu không
        \end{itemize}
        \item Return health response
    \end{enumerate}
\end{itemize}

\subsection{Scheduler Services}

\subsubsection{EmbeddingScheduler}

\textbf{File}: \texttt{src/services/embedding\_scheduler.py}

\textbf{Theo codebase hiện tại}, class \texttt{EmbeddingScheduler}:

\begin{itemize}
    \item \textbf{Chức năng}: Background scheduler cho periodic embedding refresh (12-hour cycle)
    \item \textbf{Initialization}:
    \begin{itemize}
        \item \texttt{refresh\_interval\_hours: float = 12.0}
        \item \texttt{batch\_size: int = 100}
        \item \texttt{max\_items\_per\_cycle: int = 10000}
        \item \texttt{running: bool = False}
    \end{itemize}
    
    \item \textbf{Method \texttt{run\_refresh\_cycle()}}:
    \begin{enumerate}
        \item Create database session nếu không có
        \item Create \texttt{OptimizedEmbeddingService}
        \item Fetch candidates needing refresh: \texttt{batch\_get\_candidates\_needing\_refresh(limit=max\_items\_per\_cycle)}
        \item Process candidates: \texttt{batch\_process\_candidates(candidates)}
        \item Fetch jobs needing refresh: \texttt{batch\_get\_jobs\_needing\_refresh(limit=max\_items\_per\_cycle)}
        \item Process jobs: \texttt{batch\_process\_jobs(jobs)}
        \item Clear expired cache: \texttt{cache.clear\_expired()}
        \item Return statistics: \texttt{\{'candidates\_processed', 'jobs\_processed', 'duration\_seconds'\}}
    \end{enumerate}
    
    \item \textbf{Method \texttt{run\_continuous()}}:
    \begin{itemize}
        \item Run scheduler continuously (for background worker)
        \item Parameter: \texttt{check\_interval\_seconds: int = 300} (5 minutes)
        \item Logic:
        \begin{itemize}
            \item Check if refresh is needed (dựa trên \texttt{last\_refresh} và \texttt{refresh\_interval\_hours})
            \item Nếu cần: run refresh cycle
            \item Sleep until next check
            \item Handle KeyboardInterrupt để stop gracefully
        \end{itemize}
    \end{itemize}
    
    \item \textbf{Method \texttt{stop()}}: Set \texttt{running = False}
    
    \item \textbf{Function \texttt{run\_scheduler\_worker()}}:
    \begin{itemize}
        \item Run scheduler as background worker
        \item Designed to run as separate process/daemon
        \item Parameters: \texttt{refresh\_interval\_hours}, \texttt{batch\_size}, \texttt{check\_interval\_seconds}
    \end{itemize}
    
    \item \textbf{Command-line interface}:
    \begin{itemize}
        \item Arguments:
        \begin{itemize}
            \item \texttt{--refresh-interval}: Refresh interval in hours (default: 12)
            \item \texttt{--batch-size}: Batch size for processing (default: 100)
            \item \texttt{--check-interval}: Check interval in seconds (default: 300)
            \item \texttt{--once}: Run once and exit (for cron jobs)
        \end{itemize}
        \item Nếu \texttt{--once}: Run one cycle và exit
        \item Nếu không: Run continuously
    \end{itemize}
\end{itemize}
